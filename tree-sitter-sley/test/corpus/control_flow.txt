module app.maps_for

task main -> Int {
  bind names = ["ada", "grace", "linus"]
  bind scores: Map<Text, Int> = map {
    "ada": 3,
    "grace": 5,
    "linus": 7,
  }
  tally total = 0
  each name in names {
    set total = total + scores[name]
  }
  if total > 0 {
    return total
  } else {
    return 0
  }
}
